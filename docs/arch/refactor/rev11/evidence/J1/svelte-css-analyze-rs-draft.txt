//! The Svelte CSS scoping analysis — a faithful port of the official
//! `svelte@5.56.10` `phases/2-analyze/css/css-analyze.js` (the zimmerframe
//! visitor walk), `phases/2-analyze/css/utils.js` (`is_global` /
//! `is_unscoped_pseudo_class` / `is_outer_global`), and `phases/css.js`
//! (`is_keyframes_node` / `remove_css_prefix`).
//!
//! This walks `verter_css_syntax::StyleSyntaxIr` — the SOLE CSS-family syntax
//! authority — directly; it builds no competing AST. `StyleSyntaxIr`'s node
//! types (`StyleRule` / `ComplexSelector` / `SelectorCompound` /
//! `SelectorComponent`) are immutable and carry no slot for Svelte's own
//! scoping facts (`is_global_block` / `is_nested` / `has_global_selectors` /
//! per-compound `is_global` / `is_global_like` / per-complex `used`), so this
//! analysis builds an out-of-band, pointer-identity-keyed side table
//! (`CssAnalysis`) instead of mutating the shared tree — see
//! `docs/arch/refactor/rev11/evidence/J1/svelte-css-grammar-convergence-design.md`
//! §3 for the design rationale (including the `&StyleSyntaxIr` vs
//! `&mut StyleSyntaxIr` deviation flagged there for review).
//!
//! The JS visitors read ancestor context through `context.path` /
//! `state.rule` / `metadata.parent_rule` object pointers; this port carries
//! the same facts on an explicit ancestor-rule frame stack (pushed before
//! each rule's block descent), preserving the official VISIT ORDER — the
//! rule-level global-block loop first, then the prelude (per relative
//! selector: the leading-combinator check, the selector metadata, the
//! argument descent; then the complex-selector placement checks + metadata),
//! then the rule facts, then the block. A validation failure surfaces as a
//! typed [`CssAnalysisError`] carrying the exact official `e.css_*` code —
//! never a panic, never a silent accept.
//!
//! A "relative selector" (the official `RelativeSelector` node: an optional
//! leading combinator plus one compound's simple selectors) has no 1:1
//! shared-CST type — `verter_css_syntax::ComplexSelector::parts()`
//! interleaves `Compound`/`Combinator` items in source order instead. This
//! module reconstructs the pairing via [`relative_steps`]: each
//! [`SelectorCompound`] pairs with the immediately PRECEDING [`Combinator`]
//! part (`None` for the first compound), matching the official
//! "the combinator joining this compound to the previous one" definition.

use rustc_hash::FxHashMap;
use verter_span::Span;

use verter_css_syntax::{
    decode_css_identifier, CombinatorKind, ComplexSelector, ComplexSelectorPart, SelectorCombinator,
    SelectorComponent, SelectorComponentKind, SelectorCompound, SelectorPseudo, StyleBlock,
    StyleDirective, StyleRule, StyleStatement, StyleSyntaxIr,
};

/// A typed CSS analysis failure: the official validation code + the byte
/// span of the offending node.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CssAnalysisError {
    /// The official validation code (`css_global_invalid_placement` /
    /// `css_nesting_selector_invalid_placement` / …).
    pub code: &'static str,
    /// The byte span of the offending node (absolute in the component
    /// source).
    pub span: Span,
}

impl CssAnalysisError {
    fn at(code: &'static str, span: Span) -> Self {
        Self { code, span }
    }
}

type AnalyzeResult = Result<(), CssAnalysisError>;

/// A local `@keyframes` name — a member of the scope-RENAME list (the
/// official `analysis.css.keyframes` entry).
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct KeyframeName {
    /// The keyframes name exactly as the official analysis records it (the
    /// raw trimmed prelude text).
    pub name: String,
    /// The span of the name token inside the at-rule prelude (the region the
    /// renderer prefixes with `<hash>-`).
    pub name_span: Span,
}

/// A `-global-`-prefixed `@keyframes` name — EXCLUDED from the rename list;
/// the renderer strips the `-global-` prefix instead.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct GlobalKeyframeName {
    /// The name WITHOUT the `-global-` prefix.
    pub name: String,
    /// The span of the FULL prefixed name token inside the at-rule prelude
    /// (the first 8 bytes are the `-global-` prefix the renderer removes).
    pub name_span: Span,
}

/// The official `Rule.metadata` analysis facts (minus the `parent_rule`
/// object back-pointer — walkers carry ancestor context instead; the stored
/// [`is_nested`](Self::is_nested) fact records parent-rule EXISTENCE), keyed
/// by [`StyleRule`] pointer identity in [`CssAnalysis::rules`].
#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
pub struct RuleFacts {
    /// Whether ANY prelude selector is global (`:global(...)`-only or
    /// global-like).
    pub has_global_selectors: bool,
    /// Whether ANY prelude selector is local (non-global).
    pub has_local_selectors: bool,
    /// Whether this rule is a `:global { … }` BLOCK (a prelude selector
    /// starting with the argument-less `:global`).
    pub is_global_block: bool,
    /// Whether this rule is nested inside another rule (the official
    /// `metadata.parent_rule !== null` fact).
    pub is_nested: bool,
}

/// The official `ComplexSelector.metadata` analysis facts (minus the `rule`
/// object back-pointer), keyed by [`ComplexSelector`] pointer identity in
/// [`CssAnalysis::complex`].
#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
pub struct ComplexFacts {
    /// Whether EVERY child relative selector is global or global-like.
    pub is_global: bool,
    /// Whether the selector is used (globals are used by definition; the
    /// selector-to-template matcher marks the matched ones).
    pub used: bool,
}

/// The official `RelativeSelector.metadata` analysis facts, keyed by
/// [`SelectorCompound`] pointer identity in [`CssAnalysis::compounds`] (see
/// the module doc for why a compound stands in for the official
/// `RelativeSelector` node).
#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
pub struct CompoundFacts {
    /// Whether the compound is `:global(...)` / `:global` per the official
    /// `is_global` rule (unscoped pseudo-classes / pseudo-elements only after
    /// the leading `:global`).
    pub is_global: bool,
    /// Whether the compound is global-LIKE (`:host` / `:root` /
    /// `::view-transition*`, or a compound following a `:global` block
    /// selector in the same complex selector).
    pub is_global_like: bool,
    /// Whether the compound matched a template element and receives the
    /// scope class (set by the selector-to-template matcher).
    pub scoped: bool,
}

/// The analysis facts extracted from one stylesheet: the keyframes/global
/// collection plus the pointer-identity-keyed selector-metadata side table.
/// See the module doc for why this is a side table rather than fields on the
/// shared `StyleSyntaxIr` node types — pointer identity is stable for the
/// lifetime of one owned, NEVER-CLONED `StyleSyntaxIr` value (its `Vec`s are
/// built once by the parse sink and never reallocated afterward); cloning
/// the tree after analysis silently invalidates every key here.
#[derive(Debug, Clone, Default, PartialEq, Eq)]
pub struct CssAnalysis {
    /// The LOCAL `@keyframes` rename list (the official
    /// `analysis.css.keyframes`), source order.
    pub keyframes: Vec<KeyframeName>,
    /// The `-global-`-prefixed `@keyframes` names (prefix-strip list), source
    /// order.
    pub global_keyframes: Vec<GlobalKeyframeName>,
    /// Whether the component's css includes GLOBAL css (the official
    /// `analysis.css.has_global`).
    pub has_global: bool,
    /// Per-[`StyleRule`] facts, keyed by `&StyleRule as *const _ as usize`.
    pub rules: FxHashMap<usize, RuleFacts>,
    /// Per-[`ComplexSelector`] facts, keyed by
    /// `&ComplexSelector as *const _ as usize`.
    pub complex: FxHashMap<usize, ComplexFacts>,
    /// Per-[`SelectorCompound`] facts, keyed by
    /// `&SelectorCompound as *const _ as usize`.
    pub compounds: FxHashMap<usize, CompoundFacts>,
}

fn rule_key(rule: &StyleRule) -> usize {
    std::ptr::from_ref(rule) as usize
}

fn complex_key(complex: &ComplexSelector) -> usize {
    std::ptr::from_ref(complex) as usize
}

fn compound_key(compound: &SelectorCompound) -> usize {
    std::ptr::from_ref(compound) as usize
}

impl CssAnalysis {
    /// The rule facts for `rule` (default/unset when the rule was never
    /// visited by the analyzer — never the case for a rule reachable from
    /// the analyzed tree).
    #[must_use]
    pub fn rule_facts(&self, rule: &StyleRule) -> RuleFacts {
        self.rules.get(&rule_key(rule)).copied().unwrap_or_default()
    }

    /// The complex-selector facts for `complex`.
    #[must_use]
    pub fn complex_facts(&self, complex: &ComplexSelector) -> ComplexFacts {
        self.complex
            .get(&complex_key(complex))
            .copied()
            .unwrap_or_default()
    }

    /// The compound (relative-selector) facts for `compound`.
    #[must_use]
    pub fn compound_facts(&self, compound: &SelectorCompound) -> CompoundFacts {
        self.compounds
            .get(&compound_key(compound))
            .copied()
            .unwrap_or_default()
    }

    fn rule_facts_mut(&mut self, rule: &StyleRule) -> &mut RuleFacts {
        self.rules.entry(rule_key(rule)).or_default()
    }

    fn complex_facts_mut(&mut self, complex: &ComplexSelector) -> &mut ComplexFacts {
        self.complex.entry(complex_key(complex)).or_default()
    }

    fn compound_facts_mut(&mut self, compound: &SelectorCompound) -> &mut CompoundFacts {
        self.compounds.entry(compound_key(compound)).or_default()
    }
}

/// One (leading combinator, compound) step of a complex selector — the
/// reconstruction of the official `RelativeSelector` node described in the
/// module doc. A dangling trailing combinator with no following compound
/// (only reachable from a malformed/incomplete selector, which
/// `validate_svelte_style_ir` already rejects before analysis runs) is
/// dropped rather than represented.
fn relative_steps(complex: &ComplexSelector) -> Vec<(Option<&SelectorCombinator>, &SelectorCompound)> {
    let mut steps = Vec::new();
    let mut pending: Option<&SelectorCombinator> = None;
    for part in complex.parts() {
        match part {
            ComplexSelectorPart::Combinator(combinator) => pending = Some(combinator),
            ComplexSelectorPart::Compound(compound) => {
                steps.push((pending.take(), compound));
            }
        }
    }
    steps
}

/// The decoded name of a pseudo-class/pseudo-element/functional-pseudo
/// component (a span-slice + the SAME shared decode function
/// `verter_css_syntax::selector` itself uses — never independent
/// classification), or `None` for a component with no name span.
fn component_name<'a>(source: &'a str, component: &SelectorComponent) -> Option<std::borrow::Cow<'a, str>> {
    let span = component.name_span()?;
    decode_css_identifier(&source[span.start as usize..span.end as usize]).ok()
}

/// Whether `component` is a Svelte-model "PseudoClass" simple selector — the
/// shared grammar splits this into `PseudoClass` (non-functional) and
/// `FunctionalPseudo` (`:is`/`:where`/`:not`/`:has`/`:global`/`:nth-child`/
/// any other functional pseudo-class), both of which the official Svelte AST
/// represents as the SAME `SimpleSelector::PseudoClass` node.
fn is_pseudo_class_component(component: &SelectorComponent) -> bool {
    matches!(
        component.kind(),
        SelectorComponentKind::PseudoClass | SelectorComponentKind::FunctionalPseudo
    )
}

/// The official `is_keyframes_node`.
pub(super) fn is_keyframes_node(source: &str, atrule: &StyleDirective) -> bool {
    remove_css_prefix(&atrule_name(source, atrule)) == "keyframes"
}

/// The official `remove_css_prefix`: strip a `-webkit-` / `-moz-` / `-o-` /
/// `-ms-` browser prefix.
pub(super) fn remove_css_prefix(name: &str) -> &str {
    for prefix in ["-webkit-", "-moz-", "-o-", "-ms-"] {
        if let Some(stripped) = name.strip_prefix(prefix) {
            return stripped;
        }
    }
    name
}

/// The at-rule NAME span (the official `Atrule.name_span`): `head_span`
/// covers the whole `@ident` token INCLUDING the `@`, so the name span is
/// `head_span` minus its leading byte.
pub(super) fn atrule_name_span(atrule: &StyleDirective) -> Span {
    let head = atrule.head_span();
    Span::new(head.start.saturating_add(1), head.end)
}

/// The at-rule's decoded NAME (the official `Atrule.name`).
pub(super) fn atrule_name<'a>(source: &'a str, atrule: &StyleDirective) -> std::borrow::Cow<'a, str> {
    let span = atrule_name_span(atrule);
    decode_css_identifier(&source[span.start as usize..span.end as usize])
        .unwrap_or(std::borrow::Cow::Borrowed(""))
}

/// The at-rule's TRIMMED prelude text (the official `Atrule.prelude` —
/// `read_value`'s trimmed result). `opaque_args().span()` is byte-identical
/// to the official `prelude_span` (both run from right after the name token
/// to right before the block/terminator), so trimming its raw source text
/// with the SAME JS-whitespace rule `read_value` itself uses reproduces the
/// official trimmed prelude without a second parse.
pub(super) fn atrule_prelude<'a>(source: &'a str, atrule: &StyleDirective) -> &'a str {
    let span = atrule.opaque_args().span();
    verter_css_syntax::svelte_trim_js_whitespace(&source[span.start as usize..span.end as usize])
}

/// The span of the keyframes NAME token inside the at-rule prelude — the
/// official renderer's scan (skip the spaces after the at-rule name, then
/// collect to the first `{` or space), anchored on the parsed name span
/// (positionally exact even for an escaped at-rule name). A post-parse
/// POSITIONAL scan over already-delimited raw source, not new CSS
/// classification.
pub(super) fn keyframes_name_token_span(source: &str, atrule: &StyleDirective) -> Span {
    let bytes = source.as_bytes();
    let mut start = atrule_name_span(atrule).end as usize;
    while bytes.get(start) == Some(&b' ') {
        start += 1;
    }
    let mut end = start;
    while end < bytes.len() && bytes[end] != b'{' && bytes[end] != b' ' {
        end += 1;
    }
    Span::new(start as u32, end as u32)
}

/// One ancestor-rule frame — the facts the official visitors read through
/// `metadata.parent_rule` / `context.path`, computed BEFORE descending into
/// the rule's block (matching the official visit order, where a rule's
/// prelude metadata is complete before its block is visited).
struct RuleFrame {
    /// `metadata.is_global_block`.
    is_global_block: bool,
    /// Whether the frame's rule itself has a parent rule
    /// (`metadata.parent_rule !== null` one level up).
    is_nested: bool,
    /// `metadata.has_global_selectors`.
    has_global_selectors: bool,
    /// `prelude.children[0].children.length === 1 &&
    /// prelude.children[0].children[0].selectors.length === 1` — the
    /// lone-simple first selector shape the nesting-in-global-block check
    /// reads.
    prelude_first_is_lone_simple: bool,
    /// `prelude.children.some(c => c.children.length === 1 &&
    /// c.children[0].metadata.is_global)` — the nesting-scope used-marking's
    /// `parent_is_global`.
    prelude_has_lone_global: bool,
}

/// The facts of the rule whose prelude is CURRENTLY being visited (the
/// official `context.state.rule`), needed by the nesting-selector placement
/// check.
struct OwnRuleFacts {
    /// Whether the rule has a parent rule.
    is_nested: bool,
    /// The span of the one nesting selector a TOP-LEVEL rule may legally
    /// carry — the `:global(&)` form's inner `&` — when the prelude has that
    /// exact lone shape.
    allowed_nesting_span: Option<Span>,
}

/// Analyze `tree` (already parsed through the shared `StyleSyntaxIr`
/// authority): validate `:global` / nesting placement, build the selector
/// metadata side table, and collect the keyframes + global facts.
pub fn analyze_stylesheet(
    source: &str,
    tree: &StyleSyntaxIr,
) -> Result<CssAnalysis, CssAnalysisError> {
    let mut analyzer = Analyzer {
        source,
        analysis: CssAnalysis::default(),
    };
    let mut rules = Vec::new();
    analyzer.analyze_statements(tree.statements(), &mut rules)?;
    Ok(analyzer.analysis)
}

struct Analyzer<'src> {
    source: &'src str,
    analysis: CssAnalysis,
}

impl Analyzer<'_> {
    /// Visit a rule-position statement list (the stylesheet top level, or an
    /// at-rule/rule block).
    fn analyze_statements(
        &mut self,
        statements: &[StyleStatement],
        rules: &mut Vec<RuleFrame>,
    ) -> AnalyzeResult {
        for statement in statements {
            match statement {
                StyleStatement::Rule(rule) => self.analyze_rule(rule, rules)?,
                StyleStatement::AtRule(atrule) => self.analyze_atrule(atrule, rules)?,
                // Declarations carry no analysis. `MixinOrFunction`/`Unknown`
                // statements cannot occur under `CssDialect::Css` (Sass/Stylus
                // grammar / parse-recovery artifacts respectively) and
                // `validate_svelte_style_ir` already rejects any `Unknown`
                // reaching this point — both stay a defensive no-op rather
                // than a panic.
                StyleStatement::Declaration(_)
                | StyleStatement::MixinOrFunction(_)
                | StyleStatement::Unknown(_) => {}
            }
        }
        Ok(())
    }

    /// The official `Atrule` visitor: keyframes collection (with the
    /// `-global-` / `:global {}`-block exclusions), then the block descent.
    fn analyze_atrule(
        &mut self,
        atrule: &StyleDirective,
        rules: &mut Vec<RuleFrame>,
    ) -> AnalyzeResult {
        if is_keyframes_node(self.source, atrule) {
            let in_global_block = rules.iter().any(|frame| frame.is_global_block);
            let prelude = atrule_prelude(self.source, atrule).to_string();
            if !prelude.starts_with("-global-") && !in_global_block {
                let name_span = keyframes_name_token_span(self.source, atrule);
                self.analysis.keyframes.push(KeyframeName {
                    name: prelude,
                    name_span,
                });
            } else if prelude.starts_with("-global-") {
                // The keyframe is global even if its block is empty.
                self.analysis.has_global |= is_unscoped(rules);
                let name_span = keyframes_name_token_span(self.source, atrule);
                self.analysis.global_keyframes.push(GlobalKeyframeName {
                    name: prelude["-global-".len()..].to_string(),
                    name_span,
                });
            }
        }
        // `context.next()` — descend into the block (nested rules inside
        // `@media`, the keyframe step rules, …).
        if let Some(block) = atrule.body() {
            self.analyze_block(block, rules)?;
        }
        Ok(())
    }

    /// Visit a block's children: declarations carry no analysis; rules and
    /// at-rules recurse.
    fn analyze_block(&mut self, block: &StyleBlock, rules: &mut Vec<RuleFrame>) -> AnalyzeResult {
        self.analyze_statements(block.statements(), rules)
    }

    /// The official `Rule` visitor.
    fn analyze_rule(&mut self, rule: &StyleRule, rules: &mut Vec<RuleFrame>) -> AnalyzeResult {
        let is_nested = !rules.is_empty();
        self.analysis.rule_facts_mut(rule).is_nested = is_nested;

        // The first Declaration child (the global-block declaration check
        // target).
        let first_declaration_span = rule.body().statements().iter().find_map(|statement| {
            if let StyleStatement::Declaration(declaration) = statement {
                Some(declaration.span())
            } else {
                None
            }
        });

        let selectors = rule.selector_list().selectors();

        // ── The `:global {}`-block detection loop (official: runs BEFORE the
        // prelude visit; `:global x, :global y` is allowed because CSS
        // preprocessors generate it from `:global { x, y {...} }`). ──
        for complex in selectors {
            let mut is_global_block = false;
            let steps = relative_steps(complex);

            for (step_index, (combinator, compound)) in steps.iter().enumerate() {
                let global_idx = compound
                    .components()
                    .iter()
                    .position(|component| is_global_block_selector(self.source, component));

                if is_global_block {
                    self.analysis.compound_facts_mut(compound).is_global_like = true;
                }

                match global_idx {
                    Some(0) => {
                        let components = compound.components();
                        if components.len() > 1 && step_index == 0 && !is_nested {
                            return Err(CssAnalysisError::at(
                                "css_global_block_invalid_modifier_start",
                                components[1].span(),
                            ));
                        }
                        // `compound` starts with `:global` — a global block.
                        self.analysis.rule_facts_mut(rule).is_global_block = true;
                        is_global_block = true;

                        for extra in &components[1..] {
                            self.mark_argument_selectors_used_shallow(extra);
                        }

                        if let Some(combinator) = combinator {
                            if combinator.kind() != CombinatorKind::Descendant {
                                return Err(CssAnalysisError::at(
                                    "css_global_block_invalid_combinator",
                                    compound.span(),
                                ));
                            }
                        }

                        // just `:global`, not e.g. `:global x`.
                        let is_lone_global = steps.len() == 1 && compound.components().len() == 1;

                        if is_lone_global && selectors.len() > 1 {
                            // `:global, :global x { z { ... } }` would become
                            // `x { z { ... } }`, constraining `z` in a way the
                            // author did not intend.
                            return Err(CssAnalysisError::at(
                                "css_global_block_invalid_list",
                                rule.selector_list().span(),
                            ));
                        }

                        if let Some(declaration_span) = first_declaration_span {
                            // `:global { color: red; }` is invalid, but
                            // `foo :global { color: red; }` is valid.
                            if selectors.len() == 1 && is_lone_global {
                                return Err(CssAnalysisError::at(
                                    "css_global_block_invalid_declaration",
                                    declaration_span,
                                ));
                            }
                        }
                    }
                    Some(idx) => {
                        return Err(CssAnalysisError::at(
                            "css_global_block_invalid_modifier",
                            compound.components()[idx].span(),
                        ));
                    }
                    None => {}
                }
            }

            if self.analysis.rule_facts(rule).is_global_block && !is_global_block {
                return Err(CssAnalysisError::at(
                    "css_global_block_invalid_list",
                    rule.selector_list().span(),
                ));
            }
        }

        // ── The prelude visit (SelectorList → ComplexSelector →
        // RelativeSelector, with `state.rule` = this rule). ──
        let own = OwnRuleFacts {
            is_nested,
            allowed_nesting_span: allowed_top_level_nesting_span(self.source, rule),
        };
        for complex in selectors {
            self.analyze_complex_selector(complex, rules, &own, false)?;
        }

        // ── The post-prelude rule facts. ──
        for complex in selectors {
            let is_global = self.analysis.complex_facts(complex).is_global;
            let facts = self.analysis.rule_facts_mut(rule);
            facts.has_global_selectors |= is_global;
            facts.has_local_selectors |= !is_global;
        }

        // If this rule has a ComplexSelector whose RelativeSelector children
        // are all `:global(...)`, and the rule contains DECLARATIONS (rather
        // than just nested rules), the component as a whole includes global
        // CSS.
        let declaration_count = rule
            .body()
            .statements()
            .iter()
            .filter(|statement| matches!(statement, StyleStatement::Declaration(_)))
            .count();
        let rule_facts = self.analysis.rule_facts(rule);
        self.analysis.has_global |=
            rule_facts.has_global_selectors && declaration_count > 0 && is_unscoped(rules);

        // ── The block visit, with this rule's frame pushed. ──
        rules.push(RuleFrame {
            is_global_block: rule_facts.is_global_block,
            is_nested,
            has_global_selectors: rule_facts.has_global_selectors,
            prelude_first_is_lone_simple: selectors.first().is_some_and(|complex| {
                let steps = relative_steps(complex);
                steps.len() == 1 && steps[0].1.components().len() == 1
            }),
            prelude_has_lone_global: selectors.iter().any(|complex| {
                let steps = relative_steps(complex);
                steps.len() == 1 && self.analysis.compound_facts(steps[0].1).is_global
            }),
        });
        let result = self.analyze_block(rule.body(), rules);
        rules.pop();
        result
    }

    /// The official `ComplexSelector` visitor (children first — the
    /// `context.next()` head call — then the `:global` placement checks, then
    /// the metadata).
    fn analyze_complex_selector(
        &mut self,
        complex: &ComplexSelector,
        rules: &[RuleFrame],
        own: &OwnRuleFacts,
        in_pseudo_args: bool,
    ) -> AnalyzeResult {
        let steps = relative_steps(complex);

        // `context.next()` — the RelativeSelector children.
        for (index, (_, compound)) in steps.iter().enumerate() {
            self.analyze_relative_selector(compound, index, rules, own, in_pseudo_args)?;
        }

        // ── `:global` placement over the whole complex selector. ──
        if let Some(global_idx) = steps
            .iter()
            .position(|(_, compound)| is_global(self.source, compound))
        {
            let global_compound = steps[global_idx].1;
            let first = global_compound.components()[0];
            let first_span = first.span();
            let first_args_present = is_pseudo_class_component(first)
                && first.pseudo().and_then(SelectorPseudo::selector_list).is_some();
            // A block-form `:global` nested inside a pseudo-class is invalid.
            if in_pseudo_args && !first_args_present {
                return Err(CssAnalysisError::at(
                    "css_global_block_invalid_placement",
                    first_span,
                ));
            }
            // `:global(...)` must not sit in the MIDDLE of a selector
            // (multiple `:global(...)` in sequence are ok).
            if first_args_present && global_idx != 0 && global_idx != steps.len() - 1 {
                for (_, compound) in &steps[(global_idx + 1)..] {
                    if !is_global(self.source, compound) {
                        return Err(CssAnalysisError::at(
                            "css_global_invalid_placement",
                            first_span,
                        ));
                    }
                }
            }
        }

        // ── `:global(...)` must not lead to invalid css once removed. ──
        for (_, compound) in &steps {
            let components = compound.components();
            for (i, component) in components.iter().enumerate() {
                if !is_pseudo_class_component(component) {
                    continue;
                }
                let Some(name) = component_name(self.source, component) else {
                    continue;
                };
                if name.as_ref() != "global" {
                    continue;
                }
                let args = component.pseudo().and_then(SelectorPseudo::selector_list);
                // `:global(element)` must be at the first position in a
                // compound selector.
                let arg_first = args
                    .and_then(|list| list.selectors().first())
                    .and_then(|first| relative_steps(first).first().copied());
                if let Some((_, arg_compound)) = arg_first {
                    let is_type_first = arg_compound
                        .components()
                        .first()
                        .is_some_and(|c| c.kind() == SelectorComponentKind::Type);
                    if is_type_first && i != 0 {
                        return Err(CssAnalysisError::at(
                            "css_global_invalid_selector_list",
                            component.span(),
                        ));
                    }
                }
                // `:global(.class)` must not be followed by a type selector
                // (`:global(.class)element`).
                if let Some(next) = components.get(i + 1) {
                    if next.kind() == SelectorComponentKind::Type {
                        return Err(CssAnalysisError::at(
                            "css_type_selector_invalid_placement",
                            next.span(),
                        ));
                    }
                }
                // `:global(...)` must contain a single selector (a standalone
                // `:global()` with multiple selectors is OK).
                if let Some(list) = args {
                    if list.selectors().len() > 1 && (steps.len() > 1 || components.len() > 1) {
                        return Err(CssAnalysisError::at(
                            "css_global_invalid_selector",
                            component.span(),
                        ));
                    }
                }
            }
        }

        // ── The metadata. ──
        let is_global_complex = steps.iter().all(|(_, compound)| {
            let facts = self.analysis.compound_facts(compound);
            facts.is_global || facts.is_global_like
        });
        {
            let facts = self.analysis.complex_facts_mut(complex);
            facts.is_global = is_global_complex;
            facts.used |= is_global_complex;
        }

        // Mark `&:hover` in `:global(.foo) { &:hover { color: green } }` as
        // used.
        if own.is_nested {
            let leads_with_nesting = steps
                .first()
                .and_then(|(_, compound)| compound.components().first())
                .is_some_and(|component| component.kind() == SelectorComponentKind::Nesting);
            if leads_with_nesting {
                let first_selectors = steps[0].1.components();
                let no_nesting_scope = match first_selectors.get(1) {
                    Some(component) if is_pseudo_class_component(component) => {
                        is_unscoped_pseudo_class(self.source, component)
                    }
                    _ => true,
                };
                let parent_is_global = rules
                    .last()
                    .is_some_and(|frame| frame.prelude_has_lone_global);
                if no_nesting_scope && parent_is_global {
                    self.analysis.complex_facts_mut(complex).used = true;
                }
            }
        }

        Ok(())
    }

    /// The official `RelativeSelector` visitor (the leading-combinator check,
    /// the `is_global` / `is_global_like` metadata, the used-marking, then
    /// the `context.next()` descent into pseudo-class arguments + nesting
    /// selectors). `compound` is the relative selector's compound; its
    /// leading combinator (if any) was already read by the caller from
    /// [`relative_steps`].
    fn analyze_relative_selector(
        &mut self,
        compound: &SelectorCompound,
        index: usize,
        rules: &[RuleFrame],
        own: &OwnRuleFacts,
        in_pseudo_args: bool,
    ) -> AnalyzeResult {
        // NOTE: the leading-combinator INVALID-PLACEMENT check
        // (`css_selector_invalid` on a non-nested top-level first compound)
        // is performed by the caller ([`Self::analyze_complex_selector`])
        // reading `relative_steps`' combinator directly — kept there because
        // `relative_steps` already computed the pairing once; duplicating it
        // here would re-walk `complex.parts()`.
        let components = compound.components();

        // ── The metadata. ──
        let compound_is_global = !components.is_empty() && is_global(self.source, compound);

        let mut global_like = self.analysis.compound_facts(compound).is_global_like;
        if !components.is_empty()
            && components
                .iter()
                .all(|component| is_pseudo_class_component(component) || component.kind() == SelectorComponentKind::PseudoElement)
        {
            global_like |= match components[0].kind() {
                SelectorComponentKind::PseudoClass | SelectorComponentKind::FunctionalPseudo => {
                    component_name(self.source, &components[0]).is_some_and(|name| name.as_ref() == "host")
                }
                SelectorComponentKind::PseudoElement => component_name(self.source, &components[0])
                    .is_some_and(|name| {
                        matches!(
                            name.as_ref(),
                            "view-transition"
                                | "view-transition-group"
                                | "view-transition-old"
                                | "view-transition-new"
                                | "view-transition-image-pair"
                        )
                    }),
                _ => false,
            };
        }

        // `:root.y:has(.x)` is not global: while `.y` is unscoped, the `.x`
        // inside `:has(...)` should stay scoped.
        let has_root = components.iter().any(|component| {
            is_pseudo_class_component(component)
                && component_name(self.source, component).is_some_and(|name| name.as_ref() == "root")
        });
        let has_has = components.iter().any(|component| {
            is_pseudo_class_component(component)
                && component_name(self.source, component).is_some_and(|name| name.as_ref() == "has")
        });
        global_like |= has_root && !has_has;

        {
            let facts = self.analysis.compound_facts_mut(compound);
            facts.is_global = compound_is_global;
            facts.is_global_like = global_like;
        }

        if compound_is_global || global_like {
            // So that nested selectors like `:root:not(.x)` are not marked as
            // unused.
            for component in components {
                self.mark_argument_selectors_used_recursive(component);
            }
        }

        // `context.next()` — the simple-selector children: nesting-selector
        // placement + the pseudo-class argument selector lists.
        for component in components {
            match component.kind() {
                SelectorComponentKind::Nesting => {
                    self.check_nesting_selector_placement(component.span(), rules, own)?;
                }
                _ if is_pseudo_class_component(component) => {
                    let Some(args) = component.pseudo().and_then(SelectorPseudo::selector_list)
                    else {
                        continue;
                    };
                    for inner in args.selectors() {
                        self.analyze_complex_selector(inner, rules, own, true)?;
                    }
                }
                _ => {}
            }
        }

        let _ = index;
        Ok(())
    }

    /// The official `NestingSelector` visitor: a `&` is legal inside a nested
    /// rule; at the top level only as the lone `:global(&)`; inside a lone
    /// `:global {}` block a `&` modifier is invalid.
    fn check_nesting_selector_placement(
        &mut self,
        span: Span,
        rules: &[RuleFrame],
        own: &OwnRuleFacts,
    ) -> AnalyzeResult {
        match rules.last() {
            None => {
                // No parent rule: the one legal form is the lone `:global(&)`
                // (see MDN's "using `&` outside a nested rule").
                if own.allowed_nesting_span != Some(span) {
                    return Err(CssAnalysisError::at(
                        "css_nesting_selector_invalid_placement",
                        span,
                    ));
                }
            }
            Some(parent) => {
                // `:global { &.foo { ... } }` is invalid.
                if parent.is_global_block
                    && !parent.is_nested
                    && parent.prelude_first_is_lone_simple
                {
                    return Err(CssAnalysisError::at(
                        "css_global_block_invalid_modifier_start",
                        span,
                    ));
                }
            }
        }
        Ok(())
    }

    /// The rule-loop used-marking (`walk(selector, { ComplexSelector(node) {
    /// node.metadata.used = true } })` — the visitor does NOT continue the
    /// walk, so only the OUTERMOST argument selectors are marked).
    fn mark_argument_selectors_used_shallow(&mut self, component: &SelectorComponent) {
        if !is_pseudo_class_component(component) {
            return;
        }
        if let Some(args) = component.pseudo().and_then(SelectorPseudo::selector_list) {
            for complex in args.selectors() {
                self.analysis.complex_facts_mut(complex).used = true;
            }
        }
    }

    /// The relative-selector used-marking (`walk(child, { ComplexSelector(node,
    /// context) { node.metadata.used = true; context.next(); } })` — the
    /// visitor CONTINUES the walk, marking every nested argument selector).
    fn mark_argument_selectors_used_recursive(&mut self, component: &SelectorComponent) {
        if !is_pseudo_class_component(component) {
            return;
        }
        let Some(args) = component.pseudo().and_then(SelectorPseudo::selector_list) else {
            return;
        };
        for complex in args.selectors() {
            self.analysis.complex_facts_mut(complex).used = true;
            for (_, compound) in relative_steps(complex) {
                for inner in compound.components() {
                    self.mark_argument_selectors_used_recursive(inner);
                }
            }
        }
    }
}

// ── the `css-analyze.js` local helpers ───────────────────────────────────────

/// The official `is_global_block_selector`: an ARGUMENT-LESS `:global`.
fn is_global_block_selector(source: &str, component: &SelectorComponent) -> bool {
    is_pseudo_class_component(component)
        && component_name(source, component).is_some_and(|name| name.as_ref() == "global")
        && component
            .pseudo()
            .and_then(SelectorPseudo::selector_list)
            .is_none()
}

/// The official `is_unscoped` over the ancestor-rule path: every ancestor
/// rule has global selectors (vacuously true at the top level).
fn is_unscoped(rules: &[RuleFrame]) -> bool {
    rules.iter().all(|frame| frame.has_global_selectors)
}

/// The span of the one legal top-level nesting selector — the inner `&` of a
/// lone `:global(&)` prelude — or `None` when the prelude has any other
/// shape.
fn allowed_top_level_nesting_span(source: &str, rule: &StyleRule) -> Option<Span> {
    let selectors = rule.selector_list().selectors();
    if selectors.len() != 1 {
        return None;
    }
    let steps = relative_steps(&selectors[0]);
    if steps.len() != 1 {
        return None;
    }
    let components = steps[0].1.components();
    if components.len() != 1 {
        return None;
    }
    let component = &components[0];
    if !is_pseudo_class_component(component) {
        return None;
    }
    if component_name(source, component).as_deref() != Some("global") {
        return None;
    }
    let args = component.pseudo().and_then(SelectorPseudo::selector_list)?;
    let first_complex = args.selectors().first()?;
    let first_step = relative_steps(first_complex);
    let first_compound = first_step.first()?.1;
    Some(first_compound.components().first()?.span())
}

// ── the `css/utils.js` ports ─────────────────────────────────────────────────

/// The official `is_global`: `:global(...)` or `:global` first, and no
/// SCOPED pseudo-class after it (only unscoped pseudo-classes /
/// pseudo-elements keep the whole compound global — `:global(button).x` is
/// still scoped because of the `.x`).
pub fn is_global(source: &str, compound: &SelectorCompound) -> bool {
    let components = compound.components();
    let Some(first) = components.first() else {
        return false;
    };
    if !is_pseudo_class_component(first) {
        return false;
    }
    let Some(name) = component_name(source, first) else {
        return false;
    };
    if name.as_ref() != "global" {
        return false;
    }
    let args_present = first.pseudo().and_then(SelectorPseudo::selector_list).is_some();
    !args_present
        || components.iter().all(|component| {
            is_unscoped_pseudo_class(source, component)
                || component.kind() == SelectorComponentKind::PseudoElement
        })
}

/// The official `is_unscoped_pseudo_class`: a pseudo-class that cannot be
/// (or is not) scoped. `:has` / `:is` / `:where` make the selector scoped;
/// `:not` stays unscoped unless an argument contains a multi-compound
/// selector; any of them is unscoped when every argument selector is global.
pub fn is_unscoped_pseudo_class(source: &str, component: &SelectorComponent) -> bool {
    if !is_pseudo_class_component(component) {
        return false;
    }
    let Some(name) = component_name(source, component) else {
        return false;
    };
    let args = component.pseudo().and_then(SelectorPseudo::selector_list);
    let scoped_family_ok = name.as_ref() != "has"
        && name.as_ref() != "is"
        && name.as_ref() != "where"
        // `:not` inverses the result, so it stays unscoped — except with a
        // multi-compound argument (`:not(.x .y)`), where `.x`/`.y` should be
        // scoped.
        && (name.as_ref() != "not"
            || args.is_none()
            || args.is_some_and(|list| {
                list.selectors()
                    .iter()
                    .all(|complex| relative_steps(complex).len() == 1)
            }));
    scoped_family_ok
        || args.is_none()
        || args.is_some_and(|list| {
            list.selectors().iter().all(|complex| {
                relative_steps(complex)
                    .iter()
                    .all(|(_, compound)| is_global(source, compound))
            })
        })
}

/// The official `is_outer_global`: `:global(...)` or `:global` first,
/// irrespective of scoped pseudo-classes after it — `:global(x):has(y)` is
/// outer-global but NOT `is_global`. Part of the faithful `css/utils.js`
/// port; its production consumer is the selector-to-template matcher
/// (`css-prune.js`'s `apply_selector`), which skips the `scoped` mark on an
/// outer-global compound when walking a complex selector against the
/// template.
pub fn is_outer_global(source: &str, compound: &SelectorCompound) -> bool {
    let components = compound.components();
    let Some(first) = components.first() else {
        return false;
    };
    if !is_pseudo_class_component(first) {
        return false;
    }
    let Some(name) = component_name(source, first) else {
        return false;
    };
    if name.as_ref() != "global" {
        return false;
    }
    let args_present = first.pseudo().and_then(SelectorPseudo::selector_list).is_some();
    !args_present
        || components.iter().all(|component| {
            is_pseudo_class_component(component)
                || component.kind() == SelectorComponentKind::PseudoElement
        })
}
